#!/bin/bash

dnf update -y
dnf install -y python3

mkdir -p /opt/retailedge

cat > /opt/retailedge/app.py <<'PYEOF'
from http.server import BaseHTTPRequestHandler, HTTPServer

class Handler(BaseHTTPRequestHandler):
    def do_GET(self):
        if self.path == "/health":
            body = b"OK"
        else:
            body = b"RetailEdge Application is running"

        self.send_response(200)
        self.send_header("Content-Type", "text/plain")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def log_message(self, format, *args):
        print(format % args)

server = HTTPServer(("0.0.0.0", 8080), Handler)
print("RetailEdge app listening on port 8080")
server.serve_forever()
PYEOF

cat > /etc/systemd/system/retailedge.service <<'SERVICEEOF'
[Unit]
Description=RetailEdge Application
After=network.target

[Service]
Type=simple
User=root
WorkingDirectory=/opt/retailedge
ExecStart=/usr/bin/python3 /opt/retailedge/app.py
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target
SERVICEEOF

systemctl daemon-reload
systemctl enable retailedge
systemctl start retailedge
